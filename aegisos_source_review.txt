def save_events(events):
    conn=connect(); ids=[]
    for e in events:
        cur=conn.execute('''INSERT INTO events(timestamp,event_type,user,source_ip,destination_ip,device,country,action,resource,bytes_out,success,metadata) VALUES(?,?,?,?,?,?,?,?,?,?,?,?)''',
        (e['timestamp'],e['event_type'],e.get('user'),e.get('source_ip'),e.get('destination_ip'),e.get('device'),e.get('country'),e.get('action'),e.get('resource'),e.get('bytes_out',0),int(e.get('success',True)),json.dumps(e.get('metadata',{}))))
        ids.append(cur.lastrowid)
    conn.commit(); conn.close()
    for e,i in zip(events,ids): e['id']=i
    return events

def process(events):
    alerts=detect(events)
    anomaly=anomaly_score(events)
    incidents=correlate(events,alerts)
    created=[]
    conn=connect()
    for group in incidents:
        risk,severity,reasons=score(group['alerts'],anomaly)
        key=f"{group['user']}|{group['source_ip']}|{datetime.now().strftime('%Y%m%d%H%M')}"
        summary=investigate(group,events,reasons)['bluf']
        mitre=[]
        types={a['type'] for a in group['alerts']}
        if 'BRUTE_FORCE' in types: mitre.append('T1110 - Brute Force')
        if 'SUSPICIOUS_LOGIN' in types: mitre.append('T1078 - Valid Accounts')
        if 'PRIVILEGE_ESCALATION' in types: mitre.append('T1068 - Exploitation for Privilege Escalation')
        if 'POSSIBLE_EXFILTRATION' in types: mitre.append('T1041 - Exfiltration Over C2 Channel')
        try:
            cur=conn.execute('''INSERT INTO incidents(incident_key,title,severity,risk_score,status,user,source_ip,summary,mitre,created_at) VALUES(?,?,?,?,?,?,?,?,?,?)''',
            (key,'; '.join(sorted(types)),severity,risk,'OPEN',group['user'],group['source_ip'],summary,json.dumps(mitre),datetime.now(timezone.utc).isoformat()))
            iid=cur.lastrowid
            for eid in group['event_ids']:
                conn.execute('INSERT OR IGNORE INTO incident_events(incident_id,event_id) VALUES(?,?)',(iid,eid))
            created.append({'id':iid,'title':'; '.join(sorted(types)),'severity':severity,'risk_score':risk,'status':'OPEN','user':group['user'],'source_ip':group['source_ip'],'summary':summary,'mitre':mitre,'reasons':reasons})
        except Exception:
            pass
    conn.commit(); conn.close()
    return {'events_created':len(events),'alerts':alerts,'anomaly_score':anomaly,'incidents':created}

import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parents[2] / "data" / "aegisos.db"
DB_PATH.parent.mkdir(parents=True, exist_ok=True)


def connect():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = connect()
    conn.executescript('''
    CREATE TABLE IF NOT EXISTS events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        timestamp TEXT NOT NULL,
        event_type TEXT NOT NULL,
        user TEXT,
        source_ip TEXT,
        destination_ip TEXT,
        device TEXT,
        country TEXT,
        action TEXT,
        resource TEXT,
        bytes_out INTEGER DEFAULT 0,
        success INTEGER DEFAULT 1,
        metadata TEXT DEFAULT '{}'
    );
    CREATE TABLE IF NOT EXISTS incidents (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        incident_key TEXT UNIQUE NOT NULL,
        title TEXT NOT NULL,
        severity TEXT NOT NULL,
        risk_score INTEGER NOT NULL,
        status TEXT NOT NULL DEFAULT 'OPEN',
        user TEXT,
        source_ip TEXT,
        summary TEXT,
        mitre TEXT DEFAULT '[]',
        created_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS incident_events (
        incident_id INTEGER,
        event_id INTEGER,
        PRIMARY KEY (incident_id,event_id)
    );
    ''')
    conn.commit()
    conn.close()

WEIGHTS = {
    'BRUTE_FORCE':15,'SUSPICIOUS_LOGIN':20,'PRIVILEGE_ESCALATION':25,
    'SENSITIVE_ACCESS':20,'POSSIBLE_EXFILTRATION':30,'SUSPICIOUS_EXECUTION':20
}

def score(alerts, anomaly=0):
    reasons=[]
    total=0
    seen=set()
    for a in alerts:
        t=a['type']
        if t not in seen:
            w=WEIGHTS.get(t,10); total += w; seen.add(t)
            reasons.append({'label':a['reason'],'points':w})
    if anomaly > 0.7:
        total += 10
        reasons.append({'label':'Behavioral anomaly score exceeded 0.70','points':10})
    total=min(100,total)
    severity='CRITICAL' if total>=80 else 'HIGH' if total>=60 else 'MEDIUM' if total>=30 else 'LOW'
    return total,severity,reasons

import numpy as np
from sklearn.ensemble import IsolationForest

def anomaly_score(events):
    if not events: return 0.0
    X=[]
    for e in events:
        X.append([
            1 if e.get('event_type')=='LOGIN_FAILED' else 0,
            1 if e.get('event_type')=='PRIVILEGE_CHANGE' else 0,
            1 if e.get('event_type') in ('SENSITIVE_FILE_ACCESS','DATA_TRANSFER') else 0,
            min(int(e.get('bytes_out',0))/1_000_000_000, 10),
            1 if e.get('country') in {'Russia','China','North Korea'} else 0
        ])
    if len(X)<10: return 0.0
    model=IsolationForest(contamination='auto', random_state=42)
    pred=model.fit_predict(np.array(X))
    return round(float(np.mean(pred==-1)),3)

def investigate(incident, events, reasons):
    user=incident.get('user','unknown'); ip=incident.get('source_ip','unknown')
    types=[a['type'] for a in incident.get('alerts',[])]
    title='Possible security incident'
    if 'BRUTE_FORCE' in types and 'PRIVILEGE_ESCALATION' in types: title='Possible account takeover with privilege escalation'
    elif 'POSSIBLE_EXFILTRATION' in types: title='Possible data exfiltration'
    elif 'BRUTE_FORCE' in types: title='Possible brute-force attack'
    elif 'PRIVILEGE_ESCALATION' in types: title='Possible privilege escalation'
    summary=f'{title} involving user {user} and source {ip}. AegisOS correlated multiple security signals instead of treating them as isolated alerts.'
    evidence=[a['reason'] for a in incident.get('alerts',[])]
    recommendations=['Review the affected account and recent authentication history','Revoke active sessions if compromise is suspected','Force a password reset for the affected user','Investigate the source IP and related devices']
    if 'POSSIBLE_EXFILTRATION' in types: recommendations.insert(0,'Preserve outbound-transfer evidence and review accessed data')
    return {'title':title,'bluf':summary,'evidence':evidence,'recommendations':recommendations,'confidence':min(98,70+len(types)*7)}
